cmake_minimum_required(VERSION 3.21)
project(wb_i18n_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
    add_compile_options(/utf-8)
endif()
find_package(aria CONFIG REQUIRED COMPONENTS core binding i18n)
add_executable(wb_i18n_tests main.cpp ../../core/infra/i18n/I18n.cpp)
target_include_directories(wb_i18n_tests PRIVATE ../../core)
target_link_libraries(wb_i18n_tests PRIVATE aria::i18n aria::binding)
enable_testing()
add_test(NAME tools.i18n COMMAND wb_i18n_tests "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
